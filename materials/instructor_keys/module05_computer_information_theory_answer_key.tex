\modulekey{5}{Basic Computer and Information Theory}
\textbf{Prequiz}
\begin{enumerate}
\item
\[
13=8+4+1\Rightarrow \boxed{1101_2}.
\]
Also,
\[
10110_2=16+4+2=\boxed{22}.
\]
\item
\[
0.101_2=\frac12+\frac18=\frac58=\boxed{0.625}.
\]
\(3/16\) terminates because \(16=2^4\). \(3/10\) does not terminate because the reduced denominator has a factor of \(5\).
\item Data \(1011\) has three \(1\)s, so parity bit \(\boxed1\) makes four. Packet \(10110\) has three \(1\)s, so an even-parity receiver detects an error.
\item
\[
\text{HHHHHTTTTT}\to\boxed{5\text{H}5\text{T}}.
\]
The string XYZXYZXYZ has almost no consecutive repetition, so run-length encoding does not shorten it.
\item
\[
\frac1{32}=2^{-5}\Rightarrow I=\boxed{5\text{ bits}}.
\]
\end{enumerate}

\textbf{Matching quiz}
\begin{enumerate}
\item
\[
22=16+4+2\Rightarrow \boxed{10110_2},\qquad 10001_2=16+1=\boxed{17}.
\]
\item
\[
0.011_2=\frac14+\frac18=\boxed{0.375}.
\]
\(5/8\) terminates; \(1/5\) does not.
\item Data \(0111\) has three \(1\)s, so parity bit \(\boxed1\). Packet \(11011\) has four \(1\)s, so it passes the even-parity check.
\item \(\text{MMMMMMMMMM}\to\boxed{10\text{M}}\), which is shorter than writing ten M's.
\item Probability \(=\boxed{1/128}\). Since \(1/128=2^{-7}\), information \(=\boxed7\) bits.
\end{enumerate}

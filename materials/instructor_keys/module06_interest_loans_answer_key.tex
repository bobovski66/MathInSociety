\modulekey{6}{Interest and Loans}
\textbf{Prequiz}
\begin{enumerate}
\item
\[
A=1500(1+0.04\cdot3)=1500(1.12)=\boxed{\$1680}.
\]
Yearly interest is \(0.04(1500)=\boxed{\$60}\).
\item
\[
A=1000(1.05)^4=\boxed{\$1215.51}.
\]
The dollar interest grows because each year's interest is computed from the current balance.
\item Doubling time \(\approx72/6=\boxed{12\text{ years}}\). Four times the original amount requires about two doublings, or \(\boxed{24\text{ years}}\).
\item Monthly rate:
\[
0.09/12=\boxed{0.0075}.
\]
Interest:
\[
3000(0.0075)=\boxed{\$22.50}.
\]
Principal reduction:
\[
150-22.50=\boxed{\$127.50}.
\]
New balance:
\[
3000+22.50-150=\boxed{\$2872.50}.
\]
\item First-month interest is the \textbf{same} because the starting balance and rate are the same. Loan B, with the \(\$250\) payment, will generally be paid off sooner because each payment removes more principal, which also reduces later interest.
\end{enumerate}

\textbf{Matching quiz}
\begin{enumerate}
\item
\[
A=2000(1+0.03\cdot5)=\boxed{\$2300}.
\]
Yearly increase \(=0.03(2000)=\boxed{\$60}\).
\item
\[
500(1.08)^3=\boxed{\$629.86}.
\]
\item Doubling time \(=72/8=\boxed9\) years; two doublings \(\approx\boxed{18}\) years.
\item Monthly rate \(=0.12/12=\boxed{0.01}\). Interest \(=6000(0.01)=\boxed{\$60}\). Principal reduction \(=200-60=\boxed{\$140}\). New balance \(=6000+60-200=\boxed{\$5860}\).
\item First-month interest is the same. The \(\$250\) payment removes \(\$100\) more principal than the \(\$150\) payment because the interest charge is identical.
\end{enumerate}

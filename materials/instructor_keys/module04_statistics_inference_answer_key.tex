\modulekey{4}{Intermediate Statistics and Inference}
\textbf{Prequiz}
\begin{enumerate}
\item The interval \(165\) to \(195\) is \(\mu\pm1\sigma\), so about \(\boxed{68\%}\). The value \(210=180+2(15)\), so about \(2.5\%\) lie above it: \(\boxed{2.5\%}\). The middle \(95\%\) is \(180\pm30\), or \(\boxed{150\text{ g to }210\text{ g}}\).
\item
\[
z_A=\frac{520-500}{100}=\boxed{0.20},\qquad z_B=\frac{28-21}{5}=\boxed{1.40}.
\]
Student B has the stronger relative performance.
\item
\[
243\pm8=[\boxed{235},\boxed{251}].
\]
The claim \(230\) is not inside the interval. The value \(243\) is a \textbf{sample statistic}.
\item To halve \(M\), multiply \(n\) by \(4\):
\[
4(200)=\boxed{800}.
\]
\item
\[
z=\frac{78-60}{12}=\boxed{1.5}.
\]
Yes, it is within two standard deviations. The z-score describes an \textbf{individual observation} relative to the distribution.
\end{enumerate}

\textbf{Matching quiz}
\begin{enumerate}
\item \(19\) to \(51\) is \(35\pm16=\mu\pm2\sigma\), so about \(\boxed{95\%}\).
\item
\[
z=\frac{84-72}{6}=\boxed2.
\]
The score is above the mean.
\item
\[
47\%\pm6\%=[\boxed{41\%},\boxed{53\%}].
\]
The value \(55\%\) is not inside the interval.
\item \(31.2\) cm is a \textbf{sample statistic}; it estimates the population mean fish length.
\item Multiplying \(n\) by \(9\) multiplies \(\sqrt n\) by \(3\), so the margin of error is multiplied by \(\boxed{1/3}\).
\end{enumerate}
